cmake_minimum_required(VERSION 3.24)
project(DirectX12Course LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "DirectX 12 course requires Windows 10/11")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(../cmake/Dependencies.cmake)
course_dependencies()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

course_asset_dependencies()

# Each chapter owns its C++ and shader sources. Only third-party targets are reused.
include(CTest)
file(GLOB projects CONFIGURE_DEPENDS "lessons/*" "chapters/*")
foreach(dir IN LISTS projects)
    if(EXISTS "${dir}/CMakeLists.txt")
        add_subdirectory("${dir}")
    endif()
endforeach()
